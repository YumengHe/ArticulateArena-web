Eα split norm · One joint: E

import numpy as np
from articulatearena import (
    joint_metric,
    joint_to_endpoint_pair,
    load_joint,
    make_split_norm,
)

# The loader resolves each axis into the world frame by forward kinematics.
gt = joint_to_endpoint_pair(load_joint("ground_truth.urdf", "joint_0"))
pred = joint_to_endpoint_pair(load_joint("prediction.urdf", "joint_0"))

# alpha is an inverse length that weighs translation against rotation.
# It is injected, never hardcoded. Benchmark default: 1.0 per meter.
norm = make_split_norm(alpha=1.0)

E = joint_metric(gt.u.data, gt.v.data, pred.u.data, pred.v.data, norm)
print(E)   # 0 exactly when the two joints induce the same motion

Eα split norm · Continuous joints: Eϕ

import math
import numpy as np
from articulatearena import (
    compactified_joint_metric,
    continuous_joint_repr,
    joint_metric,
    joint_to_endpoint_pair,
    load_joint,
    make_split_norm,
    radial_compactify,
)

# The loader resolves each axis into the world frame by forward kinematics.
gt = joint_to_endpoint_pair(load_joint("ground_truth.urdf", "joint_0"))
pred = joint_to_endpoint_pair(load_joint("prediction.urdf", "joint_0"))

# alpha is an inverse length that weighs translation against rotation.
# It is injected, never hardcoded. Benchmark default: 1.0 per meter.
norm = make_split_norm(alpha=1.0)

kappa = math.pi   # compactification scale, benchmark default

# Two bounded joints: E after mapping all four endpoints into the unit ball.
E_phi = compactified_joint_metric(gt.u.data, gt.v.data, pred.u.data, pred.v.data, norm, kappa)

# A continuous joint has no finite endpoints. It enters as its antipodal
# boundary pair, and the bounded joint is compactified to meet it.
cont = load_joint("prediction.urdf", "lid")          # a continuous joint: limits are +-inf
u_c, v_c = continuous_joint_repr(cont.xi_hat.data, norm)
gt_ball = [radial_compactify(z, kappa, norm) for z in (gt.u.data, gt.v.data)]
E_phi_c = joint_metric(*gt_ball, u_c, v_c, norm)
print(E_phi, E_phi_c)   # both dimensionless, in [0, 2 * sqrt(2))

Eα split norm · Whole object: Etree

import numpy as np
from articulatearena import (
    assignment_tree_distance,
    exact_tree_distance,
    joint_metric,
    joint_to_endpoint_pair,
    load_joint,
    make_split_norm,
)

# The loader resolves each axis into the world frame by forward kinematics.
gt = joint_to_endpoint_pair(load_joint("ground_truth.urdf", "joint_0"))
pred = joint_to_endpoint_pair(load_joint("prediction.urdf", "joint_0"))

# Every moving joint of each object (fixed joints carry no motion and are left out).
gt_joints = [load_joint("ground_truth.urdf", n) for n in ("joint_0", "joint_1")]
pred_joints = [load_joint("prediction.urdf", n) for n in ("joint_0", "joint_1")]

# alpha is an inverse length that weighs translation against rotation.
# It is injected, never hardcoded. Benchmark default: 1.0 per meter.
norm = make_split_norm(alpha=1.0)

# Endpoint pairs of every moving joint, all expressed in one common frame.
to_pair = lambda j: (joint_to_endpoint_pair(j).u.data, joint_to_endpoint_pair(j).v.data)
gt_pairs = [to_pair(j) for j in gt_joints]
pred_pairs = [to_pair(j) for j in pred_joints]

# Assignment relaxation: match joints by E, an unmatched joint pays E(J, J_0),
# the cost of contracting it to the fixed joint J_0 = {0, 0}.
D = assignment_tree_distance(gt_pairs, pred_pairs, norm)

# Exact tree edit distance over (parent, child, pair) edges. `certified` is True
# when the relaxation already found an isomorphism of the trees (the common case).
# load_joint returns the joint alone, so name the links of each edge yourself
# (the parent and child attributes of the URDF <joint> elements).
gt_edges = [("base", "link_0", gt_pairs[0]), ("base", "link_1", gt_pairs[1])]
pred_edges = [("base", "link_0", pred_pairs[0]), ("base", "link_1", pred_pairs[1])]
D_exact, certified = exact_tree_distance(gt_edges, pred_edges, norm)
print(D, D_exact, certified)

EB kinetic-energy norm · One joint: E

import numpy as np
from articulatearena import (
    joint_metric,
    joint_to_endpoint_pair,
    load_joint,
    make_kinetic_energy_norm,
)
# mesh-derived inertia needs the optional extra: pip install "articulatearena[mesh]"
from articulatearena.read.inertia_loader import load_joint_child_inertia

# The loader resolves each axis into the world frame by forward kinematics.
gt = joint_to_endpoint_pair(load_joint("ground_truth.urdf", "joint_0"))
pred = joint_to_endpoint_pair(load_joint("prediction.urdf", "joint_0"))

# (m, c, I) of the ground-truth child link, computed from its meshes
# as a uniform solid. E_B then has units of meters.
link = load_joint_child_inertia("ground_truth.urdf", "joint_0")
norm = make_kinetic_energy_norm(link)

E = joint_metric(gt.u.data, gt.v.data, pred.u.data, pred.v.data, norm)
print(E)   # 0 exactly when the two joints induce the same motion
